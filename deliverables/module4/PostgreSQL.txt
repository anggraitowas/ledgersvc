Write a query that returns the Daily Total of 'amount' grouped by day:
kubectl exec -it deployment/postgres -n ledgersvc -- \
  psql -U ledger -d ledger -c "
    SELECT
      (ts AT TIME ZONE 'UTC')::date AS day,
      SUM(amount) AS daily_total
    FROM ledger_entries
    GROUP BY (ts AT TIME ZONE 'UTC')::date
    ORDER BY day;
  "
